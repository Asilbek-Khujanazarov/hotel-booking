var r7 = new Room("R7", "Standard");
var r8 = new Room("R8", "Deluxe");
var maya = new Guest("Maya", "maya@example.com");
var omar = new Guest("Omar", "omar@example.com");
var nora = new Guest("Nora", "nora@example.com");
var store = new ReservationStore();
var email = new EmailSender();
var service = new BookingService([r7, r8], store, email, args.Contains("--race") ? 500 : 0);
if (args.Contains("--race"))
{
    Console.WriteLine("Maya and Omar both book R7 for 12-14 Nov at the same moment...");
    var first = Task.Run(() => service.Reserve(maya, r7, Nov(12, 14)));
    await Task.Delay(100);
    var second = Task.Run(() => service.Reserve(omar, r7, Nov(12, 14)));
    await Task.WhenAll(first, second);
    Console.WriteLine($"Maya: {first.Result}");
    Console.WriteLine($"Omar: {second.Result}");
    var count = store.CountConfirmed(r7, Nov(12, 14));
    Console.WriteLine($"Confirmed reservations for R7 on 12-14 Nov: {count}");
    Console.WriteLine(count > 1 ? "Double booking! Both requests checked before either one saved." : "Only one guest has R7.");
}
else
{
    Console.WriteLine($"Free rooms 12-14 Nov: {Names(service.FindFreeRooms(Nov(12, 14)))}");
    Console.WriteLine($"Maya books R7 12-14 Nov: {service.Reserve(maya, r7, Nov(12, 14))}");
    Console.WriteLine($"Omar books R7 13-15 Nov: {service.Reserve(omar, r7, Nov(13, 15))}");
    Console.WriteLine($"Nora books R7 14-16 Nov: {service.Reserve(nora, r7, Nov(14, 16))}");
    Console.WriteLine($"Omar books R8 15-15 Nov: {service.Reserve(omar, r8, Nov(15, 15))}");
    Console.WriteLine($"Free rooms 12-14 Nov: {Names(service.FindFreeRooms(Nov(12, 14)))}");
    Console.WriteLine($"Maya cancels R101: {service.Cancel("R101")}");
    Console.WriteLine($"Free rooms 12-14 Nov: {Names(service.FindFreeRooms(Nov(12, 14)))}");
    Console.WriteLine($"Confirmation emails sent: {email.Sent}");
}
if (args.Contains("--verify"))
{
    static void Check(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
        Console.Error.WriteLine($"PASS: {message}");
    }
    var proofStore = new ReservationStore();
    var proofEmail = new EmailSender();
    var proofService = new BookingService([r7, r8], proofStore, proofEmail);
    proofService.Reserve(maya, r7, Nov(12, 14));
    var found = proofStore.Find("R101");
    Check(found is not null && found.Guest == maya && found.Room == r7 && found.Stay == Nov(12, 14) && found.Status == ReservationStatus.Confirmed, "R5: R101 -> Maya, R7, 12-14 November, Confirmed");
    Check(proofStore.Find("missing") is null, "Unknown id returns null");
    Check(proofService.Cancel("R101") && proofStore.Find("R101")!.Status == ReservationStatus.Cancelled, "R4: cancellation retains the record as Cancelled");
    Check(!proofService.Cancel("R101") && !proofService.Cancel("missing"), "Repeated or unknown cancellation returns False");
    Check(proofService.Reserve(omar, r8, Nov(16, 15)).Outcome == "InvalidDates", "R3: reversed dates refused");
    proofService.Reserve(maya, r8, Nov(20, 22));
    Check(proofService.Reserve(omar, r8, Nov(21, 23)).Outcome == "Unavailable", "R3: R8 shared night 21 refused");
    Check(proofService.Reserve(nora, r8, Nov(22, 24)).Outcome == "Accepted", "R3: R8 adjacent stay accepted");
}
static DateRange Nov(int start, int end) => new(new(2026, 11, start), new(2026, 11, end));
static string Names(IReadOnlyList<Room> rooms) => string.Join(", ", rooms.Select(r => r.Number));

record DateRange(DateOnly CheckIn, DateOnly CheckOut)
{
    public bool IsValid() => CheckOut > CheckIn;
    public bool Overlaps(DateRange other) => CheckIn < other.CheckOut && other.CheckIn < CheckOut;
}
record Guest(string Name, string Email);
record Room(string Number, string Type);
enum ReservationStatus { Confirmed, Cancelled }
class Reservation(string id, Guest guest, Room room, DateRange stay)
{
    public string Id { get; } = id;
    public Guest Guest { get; } = guest;
    public Room Room { get; } = room;
    public DateRange Stay { get; } = stay;
    public ReservationStatus Status { get; private set; } = ReservationStatus.Confirmed;
    public bool Blocks(Room requestedRoom, DateRange requestedStay) => Status == ReservationStatus.Confirmed && Room == requestedRoom && Stay.Overlaps(requestedStay);
    public bool Cancel()
    {
        if (Status == ReservationStatus.Cancelled) return false;
        Status = ReservationStatus.Cancelled;
        return true;
    }
}
class ReservationStore
{
    private readonly List<Reservation> rows = [];
    // Protect list operations only. HasOverlap + Save remains non-atomic on purpose.
    private readonly object gate = new();
    public int Count { get { lock (gate) return rows.Count; } }
    public bool HasOverlap(Room room, DateRange stay) { lock (gate) return rows.Any(r => r.Blocks(room, stay)); }
    public void Save(Reservation reservation) { lock (gate) rows.Add(reservation); }
    public Reservation? Find(string id) { lock (gate) return rows.Find(r => r.Id == id); }
    public int CountConfirmed(Room room, DateRange stay) { lock (gate) return rows.Count(r => r.Blocks(room, stay)); }
}
class EmailSender
{
    private int sent;
    public int Sent => sent;
    // Simulation: no real emails or network requests.
    public void SendConfirmation(Reservation reservation) => Interlocked.Increment(ref sent);
}
record BookingResult(string Outcome, string? ReservationId = null)
{
    public override string ToString() => ReservationId is null ? Outcome : $"{Outcome} {ReservationId}";
}
class BookingService(IReadOnlyList<Room> rooms, ReservationStore store, EmailSender email, int slowStepMs = 0)
{
    public IReadOnlyList<Room> FindFreeRooms(DateRange stay) => rooms.Where(room => !store.HasOverlap(room, stay)).ToArray();
    public BookingResult Reserve(Guest guest, Room room, DateRange stay)
    {
        if (!stay.IsValid()) return new("InvalidDates");
        if (store.HasOverlap(room, stay)) return new("Unavailable");
        Thread.Sleep(slowStepMs); // P03 race: another request can check before we save.
        var reservation = new Reservation($"R{101 + store.Count}", guest, room, stay);
        store.Save(reservation);
        email.SendConfirmation(reservation);
        return new("Accepted", reservation.Id);
    }
    public bool Cancel(string id) => store.Find(id)?.Cancel() ?? false;
}
